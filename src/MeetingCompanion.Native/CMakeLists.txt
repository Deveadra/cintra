cmake_minimum_required(VERSION 3.25)
project(CintraAudio LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Cintra audio requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /WX /EHsc /permissive-)
add_executable(Cintra.Audio.Native main.cpp capture.cpp)
target_link_libraries(Cintra.Audio.Native PRIVATE ole32 uuid windowsapp advapi32)
enable_testing()
add_executable(Cintra.Audio.Native.Tests ../../tests/Native/audio_tests.cpp)
target_include_directories(Cintra.Audio.Native.Tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME native-audio-offline COMMAND Cintra.Audio.Native.Tests)
add_executable(Cintra.Audio.Lifetime.Tests ../../tests/Native/lifetime_tests.cpp)
target_include_directories(Cintra.Audio.Lifetime.Tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(Cintra.Audio.Lifetime.Tests PRIVATE ole32 windowsapp)
add_test(NAME native-audio-lifetime COMMAND Cintra.Audio.Lifetime.Tests)
add_executable(Cintra.Audio.Tone ../../tests/Native/tone.cpp)
target_link_libraries(Cintra.Audio.Tone PRIVATE ole32 uuid windowsapp)
